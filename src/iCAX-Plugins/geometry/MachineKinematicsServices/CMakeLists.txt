cmake_minimum_required(VERSION 3.24)
project(iCAXMachineKinematics LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This reproducible build is MSVC x64 Debug /MDd.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDebugDLL)
set(CMAKE_CONFIGURATION_TYPES Debug CACHE STRING "" FORCE)
get_filename_component(ICAX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(ICAX_BINARY_DIRECTORY "${ICAX_ROOT}/src/x64/Debug" CACHE PATH "Existing framework import libraries")
find_package(pinocchio 2.7.0 EXACT CONFIG REQUIRED)
find_package(Ceres 2.1.0 EXACT CONFIG REQUIRED)
add_library(MachineKinematicsServices SHARED PinocchioModel.cpp CeresInverse.cpp MachineKinematicsService.cpp)
target_compile_features(MachineKinematicsServices PRIVATE cxx_std_20)
target_compile_options(MachineKinematicsServices PRIVATE /utf-8 /bigobj /wd4251 /wd4996)
target_compile_definitions(MachineKinematicsServices PRIVATE NOMINMAX BOOST_ALL_NO_LIB _ENABLE_EXTENDED_ALIGNED_STORAGE)
target_include_directories(MachineKinematicsServices PRIVATE "${ICAX_ROOT}/src/iCAX-Engine/framework" "${ICAX_ROOT}/src/iCAX-Engine/foundation")
target_link_libraries(MachineKinematicsServices PRIVATE pinocchio::pinocchio Ceres::ceres "${ICAX_BINARY_DIRECTORY}/Services.lib" "${ICAX_BINARY_DIRECTORY}/Data.lib")
enable_testing()
add_executable(MachineKinematicsNativeTests "${ICAX_ROOT}/src/tests/icax-plugins/geometry/MachineKinematicsServicesTests.cpp")
target_compile_features(MachineKinematicsNativeTests PRIVATE cxx_std_20)
target_compile_options(MachineKinematicsNativeTests PRIVATE /utf-8 /bigobj /wd4251 /wd4996)
target_compile_definitions(MachineKinematicsNativeTests PRIVATE NOMINMAX BOOST_ALL_NO_LIB ICAX_TEST_REPO_ROOT="${ICAX_ROOT}")
target_include_directories(MachineKinematicsNativeTests PRIVATE "${ICAX_ROOT}/src/iCAX-Engine/framework" "${ICAX_ROOT}/src/iCAX-Engine/foundation" "${ICAX_ROOT}/src/iCAX-Plugins/cam/Machine")
target_link_libraries(MachineKinematicsNativeTests PRIVATE MachineKinematicsServices "${ICAX_BINARY_DIRECTORY}/Machine.lib" "${ICAX_BINARY_DIRECTORY}/Data.lib" pinocchio::pinocchio)
add_test(NAME MachineKinematicsNative COMMAND MachineKinematicsNativeTests)
set_tests_properties(MachineKinematicsNative PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${ICAX_BINARY_DIRECTORY};PATH=path_list_prepend:$<TARGET_FILE_DIR:pinocchio::pinocchio>")
install(TARGETS MachineKinematicsServices RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
