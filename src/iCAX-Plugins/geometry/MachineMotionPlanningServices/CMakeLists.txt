cmake_minimum_required(VERSION 3.24)
project(iCAXMachineMotionPlanning LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDebugDLL)
set(CMAKE_CONFIGURATION_TYPES Debug CACHE STRING "" FORCE)
get_filename_component(ICAX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(ICAX_BINARY_DIRECTORY "${ICAX_ROOT}/src/x64/Debug" CACHE PATH "Existing Debug framework binaries")
set(ICAX_KINEMATICS_BUILD "${ICAX_ROOT}/output/work/machine-kinematics/plugin-build/Debug" CACHE PATH "Built Pinocchio/Ceres service")
set(ICAX_KINEMATICS_INSTALL "${ICAX_ROOT}/output/work/machine-kinematics/install" CACHE PATH "Debug dependency install")
set(ICAX_COPP_ROOT "${ICAX_ROOT}/output/work/machine-kinematics/copp/0.2.3" CACHE PATH "Pinned COPP 0.2.3 C ABI SDK")
if(NOT EXISTS "${ICAX_COPP_ROOT}/include/copp/copp.h")
  message(FATAL_ERROR "Prepare the pinned COPP SDK with src/third_party/copp/prepare-copp.ps1")
endif()
find_package(copp CONFIG REQUIRED PATHS "${ICAX_COPP_ROOT}/lib/cmake/copp" NO_DEFAULT_PATH)
add_library(MachineMotionPlanningServices SHARED MachineMotionPlanningService.cpp)
target_compile_features(MachineMotionPlanningServices PRIVATE cxx_std_20)
target_compile_options(MachineMotionPlanningServices PRIVATE /utf-8 /EHsc /GR /bigobj /wd4251 /wd4996)
target_compile_definitions(MachineMotionPlanningServices PRIVATE NOMINMAX BOOST_ALL_NO_LIB)
target_include_directories(MachineMotionPlanningServices PRIVATE "${ICAX_ROOT}/src/iCAX-Engine/framework" "${ICAX_ROOT}/src/iCAX-Engine/foundation")
target_link_libraries(MachineMotionPlanningServices PRIVATE "${ICAX_BINARY_DIRECTORY}/Services.lib" "${ICAX_BINARY_DIRECTORY}/Data.lib")
target_link_libraries(MachineMotionPlanningServices PRIVATE copp::copp)
add_custom_command(TARGET MachineMotionPlanningServices POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ICAX_COPP_ROOT}/bin/copp.dll" "$<TARGET_FILE_DIR:MachineMotionPlanningServices>")
# Compile the existing upstream-backed Descartes adapter with its explicit new
# all-waypoints contract in this isolated build; no shared build/bin is touched.
set(DESCARTES_ROOT "${ICAX_ROOT}/src/third_party/descartes-light")
add_library(DescartesPlanningServices SHARED "${ICAX_ROOT}/src/iCAX-Plugins/geometry/DescartesPlanningServices/PathOrientationPlanningService.cpp")
target_compile_features(DescartesPlanningServices PRIVATE cxx_std_20)
target_compile_options(DescartesPlanningServices PRIVATE /utf-8 /EHsc /GR /bigobj /wd4251 /wd4996 /openmp)
target_compile_definitions(DescartesPlanningServices PRIVATE NOMINMAX CONSOLE_BRIDGE_STATIC_DEFINE _ENABLE_EXTENDED_ALIGNED_STORAGE)
target_include_directories(DescartesPlanningServices PRIVATE "${ICAX_ROOT}/src/iCAX-Engine/framework" "${ICAX_ROOT}/src/iCAX-Engine/foundation" "${DESCARTES_ROOT}/install/x64/Debug/include" "${DESCARTES_ROOT}/install/x64/Debug/include/console_bridge" "${DESCARTES_ROOT}/descartes_light-49e2ee3305b5d7ba5018f28cf706e0dc71ecf6cf/descartes_light/core/include" "${DESCARTES_ROOT}/console_bridge-1.0.2/include" "${ICAX_KINEMATICS_INSTALL}/include/eigen3")
target_link_libraries(DescartesPlanningServices PRIVATE "${ICAX_BINARY_DIRECTORY}/Services.lib" "${DESCARTES_ROOT}/install/x64/Debug/lib/icax_descartes_light.lib" "${DESCARTES_ROOT}/install/x64/Debug/lib/icax_console_bridge.lib")
enable_testing()
add_executable(MachineMotionPlanningNativeTests "${ICAX_ROOT}/src/tests/icax-plugins/geometry/MachineMotionPlanningServicesTests.cpp")
target_compile_features(MachineMotionPlanningNativeTests PRIVATE cxx_std_20)
target_compile_options(MachineMotionPlanningNativeTests PRIVATE /utf-8 /EHsc /GR /bigobj /wd4251 /wd4996)
target_compile_definitions(MachineMotionPlanningNativeTests PRIVATE NOMINMAX BOOST_ALL_NO_LIB ICAX_KINEMATICS_DLL="${ICAX_KINEMATICS_BUILD}/MachineKinematicsServices.dll" ICAX_TEST_REPO_ROOT="${ICAX_ROOT}")
target_include_directories(MachineMotionPlanningNativeTests PRIVATE "${ICAX_ROOT}/src/iCAX-Engine/framework" "${ICAX_ROOT}/src/iCAX-Engine/foundation" "${ICAX_ROOT}/src/iCAX-Plugins/cam/Machine")
target_link_libraries(MachineMotionPlanningNativeTests PRIVATE MachineMotionPlanningServices "${ICAX_BINARY_DIRECTORY}/Services.lib" "${ICAX_BINARY_DIRECTORY}/Data.lib" "${ICAX_BINARY_DIRECTORY}/Machine.lib")
add_dependencies(MachineMotionPlanningNativeTests DescartesPlanningServices)
add_test(NAME MachineMotionPlanningNative COMMAND MachineMotionPlanningNativeTests)
set_tests_properties(MachineMotionPlanningNative PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${ICAX_BINARY_DIRECTORY};PATH=path_list_prepend:${ICAX_KINEMATICS_INSTALL}/bin")
install(TARGETS MachineMotionPlanningServices DescartesPlanningServices RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(FILES "${ICAX_COPP_ROOT}/bin/copp.dll" DESTINATION bin)
install(FILES "${ICAX_COPP_ROOT}/LICENSE" "${ICAX_COPP_ROOT}/icax-copp.json" "${ICAX_ROOT}/src/third_party/copp/THIRD-PARTY-NOTICES.txt" "${ICAX_ROOT}/src/third_party/copp/third-party-manifest.json" DESTINATION licenses/COPP)
