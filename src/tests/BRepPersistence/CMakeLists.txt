cmake_minimum_required(VERSION 3.24)
project(BRepPersistenceTests LANGUAGES CXX)
enable_testing()
add_executable(brep-persistence-tests BRepPersistenceTests.cpp)
target_compile_features(brep-persistence-tests PRIVATE cxx_std_20)
target_include_directories(brep-persistence-tests PRIVATE ../../iCAX-Engine/foundation)
if(MSVC)
  target_compile_options(brep-persistence-tests PRIVATE /W4 /WX /utf-8)
endif()
add_test(NAME brep-persistence COMMAND brep-persistence-tests)

if(WIN32 AND ICAX_RUNTIME_DIR AND ICAX_BREP_PLUGIN)

  set_tests_properties(brep-resource-integration PROPERTIES ENVIRONMENT "PATH=${ICAX_RUNTIME_DIR}\;$ENV{PATH}")
  if(ICAX_SOURCE_ROOT)
