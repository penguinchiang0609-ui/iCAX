# The complete pinned Sparrow kernel is an independent optional strategy.
# Its Rust state lives inside each FFI call; ordinary SheetNesting builds do
# not acquire a Rust, Boost.JSON or Sparrow dependency.
option(ICAX_SHEET_NESTING_WITH_SPARROW "Build the optional pinned Sparrow strategy" OFF)
if(ICAX_SHEET_NESTING_WITH_SPARROW)
  set(SHEET_NESTING_SPARROW_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/sparrow_backend/Cargo.toml")
  set(SHEET_NESTING_SPARROW_BACKEND_LIBRARY "" CACHE FILEPATH "Optional prebuilt pinned Sparrow backend library")
  set(SHEET_NESTING_SPARROW_CARGO_HOME "" CACHE PATH "Cargo home for the optional Sparrow build")
  set(SHEET_NESTING_SPARROW_RUSTUP_HOME "" CACHE PATH "Rustup home for the optional Sparrow build")
  if(NOT SHEET_NESTING_SPARROW_BACKEND_LIBRARY)
    find_program(SHEET_NESTING_CARGO_EXECUTABLE cargo REQUIRED)
    get_filename_component(SHEET_NESTING_RUST_BIN "${SHEET_NESTING_CARGO_EXECUTABLE}" DIRECTORY)
    find_program(SHEET_NESTING_RUSTC_EXECUTABLE rustc HINTS "${SHEET_NESTING_RUST_BIN}" REQUIRED)
    set(SHEET_NESTING_SPARROW_TARGET "${CMAKE_CURRENT_BINARY_DIR}/sparrow-target")
    if(WIN32)
      set(SHEET_NESTING_SPARROW_BACKEND_LIBRARY "${SHEET_NESTING_SPARROW_TARGET}/release/icax_sparrow_backend.lib")
    else()
      set(SHEET_NESTING_SPARROW_BACKEND_LIBRARY "${SHEET_NESTING_SPARROW_TARGET}/release/libicax_sparrow_backend.a")
    endif()
    set(SPARROW_BUILD_ENV "RUSTC=${SHEET_NESTING_RUSTC_EXECUTABLE}")
    if(SHEET_NESTING_SPARROW_CARGO_HOME)
      list(APPEND SPARROW_BUILD_ENV "CARGO_HOME=${SHEET_NESTING_SPARROW_CARGO_HOME}")
    endif()
    if(SHEET_NESTING_SPARROW_RUSTUP_HOME)
      list(APPEND SPARROW_BUILD_ENV "RUSTUP_HOME=${SHEET_NESTING_SPARROW_RUSTUP_HOME}")
    endif()
    file(GLOB_RECURSE SPARROW_RUST_SOURCES CONFIGURE_DEPENDS
      "${CMAKE_CURRENT_SOURCE_DIR}/sparrow_backend/src/*.rs"
      "${ICAX_SOURCE_ROOT}/third_party/sparrow/src/*.rs")
    add_custom_command(OUTPUT "${SHEET_NESTING_SPARROW_BACKEND_LIBRARY}"
      COMMAND ${CMAKE_COMMAND} -E env ${SPARROW_BUILD_ENV}
        "${SHEET_NESTING_CARGO_EXECUTABLE}" build --release --locked
        --manifest-path "${SHEET_NESTING_SPARROW_MANIFEST}"
        --target-dir "${SHEET_NESTING_SPARROW_TARGET}"
      DEPENDS ${SPARROW_RUST_SOURCES} "${SHEET_NESTING_SPARROW_MANIFEST}"
        "${CMAKE_CURRENT_SOURCE_DIR}/sparrow_backend/Cargo.lock"
        "${ICAX_SOURCE_ROOT}/third_party/sparrow/Cargo.toml"
      COMMENT "Building the pinned Sparrow Rust kernel" VERBATIM)
    add_custom_target(SheetNestingSparrowRustBuild DEPENDS "${SHEET_NESTING_SPARROW_BACKEND_LIBRARY}")
  endif()
  add_library(SheetNestingSparrowRust STATIC IMPORTED GLOBAL)
  set_target_properties(SheetNestingSparrowRust PROPERTIES IMPORTED_LOCATION "${SHEET_NESTING_SPARROW_BACKEND_LIBRARY}")
  if(TARGET SheetNestingSparrowRustBuild)
    add_dependencies(SheetNestingSparrowRust SheetNestingSparrowRustBuild)
  endif()
  set(SHEET_NESTING_SPARROW_BOOST_ROOT "${ICAX_SOURCE_ROOT}/third_party/coal/boost_1_87_0")
  add_library(SheetNestingSparrow STATIC NestingSparrow.cpp)
  target_include_directories(SheetNestingSparrow PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}"
    PRIVATE "${SHEET_NESTING_SPARROW_BOOST_ROOT}")
  target_link_libraries(SheetNestingSparrow PUBLIC SheetNesting PRIVATE SheetNestingSparrowRust)
  if(WIN32)
    target_link_libraries(SheetNestingSparrow PRIVATE ws2_32 userenv bcrypt ntdll advapi32)
  else()
    find_package(Threads REQUIRED)
    target_link_libraries(SheetNestingSparrow PRIVATE Threads::Threads ${CMAKE_DL_LIBS} m)
  endif()
  if(MSVC)
    target_compile_options(SheetNestingSparrow PRIVATE /W4 /utf-8 /permissive-)
  endif()
endif()
