cmake_minimum_required(VERSION 3.24)
project(TubeDesignerLicensing LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The native licensing implementation requires Windows")
endif()
add_library(td_license INTERFACE)
target_include_directories(td_license INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(td_license INTERFACE cxx_std_20)
target_compile_definitions(td_license INTERFACE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(td_license INTERFACE bcrypt ncrypt tbs crypt32 shell32 ole32)
if(MSVC)
  target_compile_options(td_license INTERFACE /W4 /WX /utf-8 /permissive-)
endif()
add_executable(td-license-probe tools/probe.cpp)
target_link_libraries(td-license-probe PRIVATE td_license)
add_executable(td-license-nv-probe tools/nv_probe.cpp)
target_link_libraries(td-license-nv-probe PRIVATE td_license)
add_executable(td-license-enrollment-probe tools/enrollment_probe.cpp)
target_link_libraries(td-license-enrollment-probe PRIVATE td_license)
add_executable(td-license-client tools/client.cpp)
target_link_libraries(td-license-client PRIVATE td_license)
add_executable(td-license-verify-ek tools/verify_ek_chain.cpp)
target_link_libraries(td-license-verify-ek PRIVATE td_license)
if(MSVC)
  target_link_options(td-license-client PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(td-license-tests tests/core_tests.cpp)
  target_link_libraries(td-license-tests PRIVATE td_license)
  add_test(NAME licensing-core COMMAND td-license-tests)
  add_executable(td-license-runtime-mode-tests tests/runtime_mode_tests.cpp)
  target_link_libraries(td-license-runtime-mode-tests PRIVATE td_license)
  add_test(NAME licensing-runtime-mode COMMAND td-license-runtime-mode-tests)
endif()
