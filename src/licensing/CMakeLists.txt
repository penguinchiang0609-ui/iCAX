cmake_minimum_required(VERSION 3.24)
project(TubeDesignerLicensing LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The native licensing implementation requires Windows")
endif()
enable_language(RC)
add_library(td_license INTERFACE)
target_include_directories(td_license INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(td_license INTERFACE cxx_std_20)
target_compile_definitions(td_license INTERFACE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(td_license INTERFACE bcrypt ncrypt tbs crypt32 shell32 ole32)
if(MSVC)
  target_compile_options(td_license INTERFACE
    "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/utf-8;/permissive->")
endif()
add_executable(td-license-probe tools/probe.cpp)
target_link_libraries(td-license-probe PRIVATE td_license)
add_executable(td-license-nv-probe tools/nv_probe.cpp)
target_link_libraries(td-license-nv-probe PRIVATE td_license)
add_executable(td-license-enrollment-probe tools/enrollment_probe.cpp)
target_link_libraries(td-license-enrollment-probe PRIVATE td_license)
add_executable(td-license-client tools/client.cpp)
target_link_libraries(td-license-client PRIVATE td_license)
add_executable(td-license-verify-ek tools/verify_ek_chain.cpp)
target_link_libraries(td-license-verify-ek PRIVATE td_license)
# One native executable; cryptography and enrollment validation run in process.
add_library(td_license_issuer_core STATIC
  issuer_native/IssuerCore.cpp
  issuer_native/IssuerCrypto.cpp
  issuer_native/IssuerSettings.cpp)
target_include_directories(td_license_issuer_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/issuer_native)
target_link_libraries(td_license_issuer_core PUBLIC td_license)
configure_file(issuer_native/IssuerResources.rc ${CMAKE_CURRENT_BINARY_DIR}/IssuerResources.rc @ONLY)
add_executable(td-license-issuer WIN32 issuer_native/IssuerWindow.cpp ${CMAKE_CURRENT_BINARY_DIR}/IssuerResources.rc)
target_link_libraries(td-license-issuer PRIVATE td_license_issuer_core
  comdlg32 comctl32 gdi32 user32)
set_target_properties(td-license-issuer PROPERTIES OUTPUT_NAME "iCAX-License-Issuer")
if(MSVC)
  set_property(TARGET td_license_issuer_core td-license-issuer
    PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_options(td-license-issuer PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
  target_link_options(td-license-issuer PRIVATE
    "/MANIFESTDEPENDENCY:type='win32' name='Microsoft.Windows.Common-Controls' version='6.0.0.0' processorArchitecture='*' publicKeyToken='6595b64144ccf1df' language='*'")
endif()
if(MSVC)
  target_link_options(td-license-client PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(td-license-issuer-tests issuer_native/IssuerCoreTests.cpp)
  target_link_libraries(td-license-issuer-tests PRIVATE td_license_issuer_core)
  if(MSVC)
    set_property(TARGET td-license-issuer-tests
      PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_test(NAME licensing-native-issuer COMMAND td-license-issuer-tests)
  add_test(NAME licensing-native-issuer-ui COMMAND td-license-issuer --self-test)
  add_executable(td-license-tests tests/core_tests.cpp)
  target_link_libraries(td-license-tests PRIVATE td_license)
  add_test(NAME licensing-core COMMAND td-license-tests)
  add_executable(td-license-upgrade-tests tests/upgrade_tests.cpp)
  target_link_libraries(td-license-upgrade-tests PRIVATE td_license)
  add_test(NAME licensing-upgrade COMMAND td-license-upgrade-tests)
  add_executable(td-license-runtime-mode-tests tests/runtime_mode_tests.cpp)
  target_link_libraries(td-license-runtime-mode-tests PRIVATE td_license)
  add_test(NAME licensing-runtime-mode COMMAND td-license-runtime-mode-tests)
  add_executable(td-license-ambiguous-mode-tests tests/runtime_mode_tests.cpp)
  target_link_libraries(td-license-ambiguous-mode-tests PRIVATE td_license)
  target_compile_definitions(td-license-ambiguous-mode-tests PRIVATE _DEBUG NDEBUG)
  set_property(TARGET td-license-ambiguous-mode-tests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebugDLL)
  add_test(NAME licensing-ambiguous-mode COMMAND td-license-ambiguous-mode-tests)
  # Explicit hardware acceptance only; never installed or linked into the client.
  add_executable(td-license-hardware-tests tests/hardware_activation_tests.cpp)
  target_link_libraries(td-license-hardware-tests PRIVATE td_license)
  # Explicit trial hardware acceptance owns and removes only its newly allocated NV index.
  add_executable(td-license-trial-hardware-tests tests/trial_hardware_tests.cpp)
  target_link_libraries(td-license-trial-hardware-tests PRIVATE td_license)
endif()

# Multi-product container tests use only ephemeral software keys and opaque
# activation/upgrade payload fixtures. No signing vault or TPM is accessed.
if(BUILD_TESTING)
  add_executable(td-license-bundle-tests tests/bundle_tests.cpp)
  target_link_libraries(td-license-bundle-tests PRIVATE td_license)
  add_test(NAME licensing-bundle COMMAND td-license-bundle-tests)
endif()
